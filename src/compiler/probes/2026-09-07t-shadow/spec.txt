// shadowing a PARAM named `x` leaves BOTH `drop x` statements resolving to the

    return m;

void MLIRGenImpl::gen_stmt_kind(lir_view::SLetView v)        {
    std::string sh_nm_(v.name());
    mlir::Value sh_prev_{};
    bool sh_had_ = false;
    if ((logos::probe::on("dropunshadow") || logos::probe::on("dropunshadow2")) &&
        !sh_nm_.empty()) {
        auto pit_ = scope_.find(sh_nm_);
        if (pit_ != scope_.end()) { sh_prev_ = pit_->second; sh_had_ = true; }
    }
    gen_let(v);
    if (sh_had_) {
        auto nit_ = scope_.find(sh_nm_);
        if (nit_ != scope_.end() && nit_->second != sh_prev_)
            logos_probe_shadow_prev()[sh_nm_] = sh_prev_;
    }
}
