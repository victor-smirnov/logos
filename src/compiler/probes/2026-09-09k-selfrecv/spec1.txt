                        // CRUDE TWIN (rule 18): the artefact escape hatch is
                        // removed outright once a `Self`-shaped trait slot is
                        // substitutable. Refuses every legitimate Self-shape
                        // disagreement too; it exists to bound the other arm.
                        if (logos::probe::on("sigselfnone")) return false;
                        // TARGETED ARM: the INDIRECTION PREFIX of the substituted
                        // trait slot vs the impl's written slot. A different
                        // prefix (`&mut S` declared, `S` written) is the receiver
                        // defect; an equal prefix over a different pointee
                        // (`&str` vs `&[u8]`) is the Self-shape artefact this
                        // hatch was written for.
                        if (logos::probe::on("sigselfdepth")) {
                            auto _pfx = [](TypeRef t) {
                                std::string s;
                                while (t && (TypeRef(t).kind() == K3::Ref ||
                                             TypeRef(t).kind() == K3::MutRef)) {
                                    s += (TypeRef(t).kind() == K3::MutRef) ? 'm' : 'r';
                                    t = TypeRef(t).pointee();
                                }
                                return s;
                            };
                            if (_pfx(sub) != _pfx(impl_t)) return false;
                        }
                        return !types_equal(sub, impl_t);
